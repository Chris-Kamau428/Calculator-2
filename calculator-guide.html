How to Make the Calculator Work

This is a separate guide. It does not change or run your calculator page. Your original index.html and script.js have been left as they were.

What the JavaScript needs

  1. Find the display and the calculator buttons in the HTML.
  2. When a button is clicked, add its value to the display's calculation.
  3. When = is clicked, parse and calculate the expression, respecting parentheses and multiplication/division order.
  4. Handle clear, delete, decimal points, and keyboard input.

Important: the display in your HTML is id="displayscreen", not id="text". The script must use the matching ID. Also, JavaScript should load after the HTML has been read; use defer on its script tag, or put the script tag just before </body>.

1. Give the buttons values and actions

Add a data-value attribute to each number and operator button. For example:

<button class="row" data-value="7">7</button>
<button class="start" data-value="+">+</button>
<button class="start" data-action="parentheses">()</button>
<button data-action="clear">C</button>
<button data-action="backspace">⌫</button>
<button data-action="equals">=</button>

Give every number, decimal point, and arithmetic operator its own data-value. Add the three action buttons for clearing, deleting the last character, and calculating. Your current HTML has no equals button, so one must be added for a clickable result.

2. Put this JavaScript in script.js

After making the HTML changes above, this is a complete example. It does not use eval(); the small parser below only accepts numbers, parentheses, and the four arithmetic operators.

"use strict";

const display = document.getElementById("displayscreen");
const buttons = document.getElementById("input");
let expression = "";

function render() {
    display.value = expression;
}

function append(value) {
    expression += value;
    render();
}

function addDecimal() {
    const currentNumber = expression.split(/[()+\-*/]/).pop();
    if (currentNumber.includes(".")) return;
    if (!currentNumber || expression.endsWith(")")) {
        append(expression.endsWith(")") ? "*0." : "0.");
    } else {
        append(".");
    }
}

function addParenthesis() {
    const opens = (expression.match(/\(/g) || []).length;
    const closes = (expression.match(/\)/g) || []).length;
    const last = expression.slice(-1);
    if (opens > closes && /[\d)]/.test(last)) {
        append(")");
    } else {
        if (/[\d)]/.test(last)) expression += "*";
        append("(");
    }
}

function calculate(source) {
    const compact = source.replace(/\s/g, "");
    const tokens = compact.match(/\d+(?:\.\d*)?|\.\d+|[()+\-*/]/g);
    if (!tokens || tokens.join("") !== compact) throw new Error("Invalid calculation");
    let position = 0;

    function primary() {
        if (tokens[position] === "(") {
            position++;
            const value = sum();
            if (tokens[position] !== ")") throw new Error("Missing )");
            position++;
            return value;
        }
        const token = tokens[position++];
        if (!token || !/^(?:\d+(?:\.\d*)?|\.\d+)$/.test(token)) throw new Error("Expected a number");
        return Number(token);
    }

    function unary() {
        if (tokens[position] === "+" || tokens[position] === "-") {
            const operator = tokens[position++];
            const value = unary();
            return operator === "-" ? -value : value;
        }
        return primary();
    }

    function product() {
        let value = unary();
        while (tokens[position] === "*" || tokens[position] === "/") {
            const operator = tokens[position++];
            const right = unary();
            if (operator === "/" && right === 0) throw new Error("Cannot divide by zero");
            value = operator === "*" ? value * right : value / right;
        }
        return value;
    }

    function sum() {
        let value = product();
        while (tokens[position] === "+" || tokens[position] === "-") {
            const operator = tokens[position++];
            const right = product();
            value = operator === "+" ? value + right : value - right;
        }
        return value;
    }

    const result = sum();
    if (position !== tokens.length || !Number.isFinite(result)) {
        throw new Error("Invalid calculation");
    }
    return Number(result.toPrecision(12)).toString();
}

buttons.addEventListener("click", (event) => {
    const button = event.target.closest("button");
    if (!button) return;

    const action = button.dataset.action;
    if (action === "clear") {
        expression = "";
        render();
    } else if (action === "backspace") {
        expression = expression.slice(0, -1);
        render();
    } else if (action === "equals") {
        try {
            expression = calculate(expression);
            render();
        } catch (error) {
            display.value = error.message === "Cannot divide by zero" ? error.message : "Error";
            expression = "";
        }
    } else if (action === "parentheses") {
        addParenthesis();
    } else if (button.dataset.value === ".") {
        addDecimal();
    } else if (button.dataset.value !== undefined) {
        append(button.dataset.value);
    }
});

document.addEventListener("keydown", (event) => {
    if (/^[0-9+\-*/()]$/.test(event.key)) {
        append(event.key);
    } else if (event.key === ".") {
        addDecimal();
    } else if (event.key === "Enter" || event.key === "=") {
        event.preventDefault();
        buttons.querySelector('[data-action="equals"]').click();
    } else if (event.key === "Backspace") {
        buttons.querySelector('[data-action="backspace"]').click();
    } else if (event.key === "Escape") {
        buttons.querySelector('[data-action="clear"]').click();
    }
});

3. Why each part matters

  • display and buttons find the matching HTML elements. The IDs must match exactly.
  • expression stores the calculation. render() copies it into the readonly display.
  • The click listener handles all buttons in one place. The button's data-value says what to append; data-action identifies special buttons.
  • calculate() reads numbers and operators in precedence order: parentheses, then multiplication/division, then addition/subtraction.
  • The keyboard listener lets you type a calculation, press Enter to get the result, Backspace to delete, or Escape to clear.

4. Load the script after the page is ready

In the page's <head>, the script tag should look like this:

<script src="script.js" defer></script>

Then open the calculator page in a browser. For example, enter 2+3*4 and press =; the result should be 14.
